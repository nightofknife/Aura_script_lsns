cmake_minimum_required(VERSION 3.24)
project(resonance_input_bridge VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The bridge requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(minhook json)
add_library(resonance_bridge_host SHARED host/host.cpp host/runtime.cpp host/il2cpp.cpp)
set_target_properties(resonance_bridge_host PROPERTIES PREFIX "")
target_link_libraries(resonance_bridge_host PRIVATE minhook nlohmann_json::nlohmann_json bcrypt advapi32)
target_include_directories(resonance_bridge_host PRIVATE "${minhook_SOURCE_DIR}/src/hde")
target_compile_definitions(resonance_bridge_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_executable(resonance_bridge_loader loader/main.cpp)
target_link_libraries(resonance_bridge_loader PRIVATE advapi32)
target_compile_definitions(resonance_bridge_loader PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MINGW)
  target_link_options(resonance_bridge_host PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(resonance_bridge_loader PRIVATE -municode -static-libgcc -static-libstdc++)
endif()
add_custom_command(TARGET resonance_bridge_host POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/profiles" "$<TARGET_FILE_DIR:resonance_bridge_host>/profiles")
install(TARGETS resonance_bridge_host resonance_bridge_loader RUNTIME DESTINATION . COMPONENT Runtime LIBRARY DESTINATION . COMPONENT Runtime)
install(DIRECTORY profiles DESTINATION . COMPONENT Runtime)
