"""Single-frame automatic native-geometry pose probe. No labels/input APIs.

Known native camera and geometry; image-oriented edge distances; automatic
multi-start SO(3) search. Scores are residuals, never probabilities.
"""
from pathlib import Path
import sys, json, time, hashlib, itertools
import cv2
import numpy as np

ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'.pytest_tmp/deep-dive-native-projection-20261022'))
from project_native import Model
from render_depth import depth_image,draw_boundary
OUT=Path(__file__).resolve().parent
IMAGE=OUT/'live-client.png'

def main():
    started=time.perf_counter();m=Model();img=cv2.imread(str(IMAGE));assert img.shape==(720,1280,3)
    gray=cv2.cvtColor(img,cv2.COLOR_BGR2GRAY);gray=cv2.GaussianBlur(gray,(3,3),.7)
    edge=cv2.Canny(gray,45,120)
    gx=cv2.Sobel(gray,cv2.CV_32F,1,0);gy=cv2.Sobel(gray,cv2.CV_32F,0,1)
    tangent=(np.arctan2(gy,gx)+np.pi/2)%np.pi
    bins=12;dist=[]
    for b in range(bins):
        delta=abs(tangent-(b+.5)*np.pi/bins);delta=np.minimum(delta,np.pi-delta)
        e=(edge>0)&(delta<np.pi/bins)
        dist.append(cv2.distanceTransform((~e).astype(np.uint8),cv2.DIST_L2,3))
    dist=np.array(dist);cv2.imwrite(str(OUT/'observed-edges.png'),edge)
    rim=np.array([c['rim'] for c in m.cells]);normal=np.array([c['normal'] for c in m.cells]);center=rim.mean(1)
    t=np.linspace(.1,.9,9);sample=rim[:,:,None,:]+(np.roll(rim,-1,axis=1)-rim)[:,:,None,:]*t[None,None,:,None]
    def score(R):
        cw=center@R.T;nw=normal@R.T;v=m.camera_C-cw
        visible=np.sum(nw*v,axis=1)/np.linalg.norm(v,axis=1)>.18
        q=m.project(rim.reshape(-1,3),R).reshape(54,4,2)
        dirs=np.roll(q,-1,axis=1)-q
        direction=(np.arctan2(dirs[:,:,1],dirs[:,:,0])%np.pi)
        bid=np.minimum((direction/np.pi*bins).astype(int),bins-1)
        p=m.project(sample.reshape(-1,3),R).reshape(54,4,9,2)
        px=np.rint(p).astype(int);inside=(px[:,:,:,0]>=1)&(px[:,:,:,0]<1279)&(px[:,:,:,1]>=90)&(px[:,:,:,1]<630)
        use=inside&visible[:,None,None]
        if use.sum()<200:return 100.
        xx=np.clip(px[:,:,:,0],0,1279);yy=np.clip(px[:,:,:,1],0,719)
        residual=dist[np.broadcast_to(bid[:,:,None],xx.shape),yy,xx]
        # Bounded robust loss limits occlusion/effect influence; no glyph centers.
        return float(np.minimum(residual[use],12).mean())
    rng=np.random.default_rng(20261022);coarse=[]
    rotations=[np.eye(3)]
    for _ in range(1536):
        q=rng.normal(size=4);q/=np.linalg.norm(q);x,y,z,w=q
        rotations.append(np.array([[1-2*(y*y+z*z),2*(x*y-z*w),2*(x*z+y*w)],[2*(x*y+z*w),1-2*(x*x+z*z),2*(y*z-x*w)],[2*(x*z-y*w),2*(y*z+x*w),1-2*(x*x+y*y)]]))
    for i,R in enumerate(rotations):coarse.append((score(R),i,R))
    coarse.sort(key=lambda x:x[0]);print('COARSE',coarse[0][0],'seconds',time.perf_counter()-started,flush=True)
    candidates=[]
    for initial,idx,R0 in coarse[:18]:
        R=R0.copy();value=initial
        for degrees in (5,2,1,.5,.2,.08):
            increments=[cv2.Rodrigues(np.eye(3)[axis]*np.deg2rad(sign*degrees))[0] for axis in range(3) for sign in (-1,1)]
            for _ in range(12):
                options=[(score(delta@R),delta@R) for delta in increments];best=min(options,key=lambda x:x[0])
                if best[0]>=value-1e-6:break
                value,R=best
        candidates.append(dict(score=value,coarse_score=initial,coarse_index=idx,rotation=R.tolist()))
    candidates.sort(key=lambda x:x['score']);chosen=candidates[0];R=np.array(chosen['rotation'])
    result=dict(mode='automatic single-frame pose probe; no manual points or old geometry used',image_sha256=hashlib.sha256(IMAGE.read_bytes()).hexdigest(),coarse_candidates=len(coarse),refined_candidates=len(candidates),elapsed_sec=time.perf_counter()-started,score_unit='clipped oriented edge distance in pixels; NOT accuracy/confidence',chosen=chosen,candidates=candidates,limitations=['single viewpoint only','cube rotational symmetry not resolved','no node/entity recognition','occlusion only checked in output proxy','no automatic production acceptance gate'])
    (OUT/'automatic-result.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
    depth=depth_image(m,R);overlay=img.copy();cells=[]
    for c in m.cells:
        if not m.visible(c,R):continue
        draw_boundary(overlay,m,c['rim'],R,depth,(0,230,0));draw_boundary(overlay,m,c['icon'],R,depth,(240,190,0))
        cells.append(dict(cell=c['key'],rim=m.project(c['rim'],R).tolist(),icon=m.project(c['icon'],R).tolist()))
    cv2.putText(overlay,'AUTOMATIC native pose | green=rim cyan=Icon | symmetry unresolved',(15,25),0,.5,(255,255,255),1,cv2.LINE_AA)
    cv2.imwrite(str(OUT/'automatic-overlay.png'),overlay)
    (OUT/'projected-cells.json').write_text(json.dumps(cells,indent=2),encoding='utf-8')
    print('RESULT',chosen['score'],'seconds',result['elapsed_sec'],flush=True)

if __name__=='__main__':
    import argparse
    parser=argparse.ArgumentParser();parser.add_argument('--image',type=Path);parser.add_argument('--output',type=Path)
    args=parser.parse_args()
    if args.image:IMAGE=args.image.resolve()
    if args.output:OUT=args.output.resolve();OUT.mkdir(parents=True,exist_ok=True)
    main()
